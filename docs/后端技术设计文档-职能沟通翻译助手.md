### 6.4 AI客户端模块

**文件**: `app/services/ai_client.py`

**职责**:
- 封装通义千问API调用
- 处理流式/非流式请求
- 统一错误处理

**实现**:

```python
import dashscope
from typing import AsyncGenerator, Optional
from app.config import settings
from app.utils.logger import logger

class QwenClient:
    """通义千问API客户端"""

    def __init__(self):
        """初始化客户端"""
        dashscope.api_key = settings.QWEN_API_KEY
        self.model = settings.QWEN_MODEL

    async def stream_generate(
        self,
        prompt: str,
        system_prompt: str = ""
    ) -> AsyncGenerator[str, None]:
        """
        流式生成内容

        Args:
            prompt: 用户提示词
            system_prompt: 系统提示词

        Yields:
            生成的文本内容

        Raises:
            Exception: API调用失败
        """
        try:
            messages = [
                {"role": "system", "content": system_prompt},
                {"role": "user", "content": prompt}
            ]

            logger.info(f"开始流式生成, 模型: {self.model}")

            response = dashscope.Generation.call(
                model=self.model,
                messages=messages,
                stream=True,
                result_format="message",
                incremental_output=True
            )

            for chunk in response:
                if chunk.status_code == 200:
                    content = chunk.output.choices[0].message.content
                    if content:
                        yield content
                else:
                    error_msg = f"API Error: {chunk.code} - {chunk.message}"
                    logger.error(error_msg)
                    raise Exception(error_msg)

        except Exception as e:
            logger.error(f"流式生成失败: {str(e)}")
            raise Exception(f"AI生成失败: {str(e)}")

    async def generate(
        self,
        prompt: str,
        system_prompt: str = ""
    ) -> str:
        """
        非流式生成内容(用于场景识别)

        Args:
            prompt: 用户提示词
            system_prompt: 系统提示词

        Returns:
            生成的完整文本

        Raises:
            Exception: API调用失败
        """
        try:
            messages = [
                {"role": "system", "content": system_prompt},
                {"role": "user", "content": prompt}
            ]

            logger.info(f"开始非流式生成, 模型: {self.model}")

            response = dashscope.Generation.call(
                model=self.model,
                messages=messages,
                result_format="message"
            )

            if response.status_code == 200:
                return response.output.choices[0].message.content
            else:
                error_msg = f"API Error: {response.code} - {response.message}"
                logger.error(error_msg)
                raise Exception(error_msg)

        except Exception as e:
            logger.error(f"非流式生成失败: {str(e)}")
            raise Exception(f"AI生成失败: {str(e)}")
```

### 6.5 翻译服务模块

**文件**: `app/services/translation.py`

**职责**:
- 翻译业务逻辑
- 提示词选择
- 流式结果返回

**实现**:

```python
from typing import AsyncGenerator
from app.services.ai_client import QwenClient
from app.prompts import product_to_dev, dev_to_product
from app.utils.logger import logger

class TranslationService:
    """翻译服务"""

    def __init__(self):
        """初始化服务"""
        self.ai_client = QwenClient()

    async def translate_stream(
        self,
        content: str,
        direction: str
    ) -> AsyncGenerator[str, None]:
        """
        流式翻译

        Args:
            content: 待翻译内容
            direction: 翻译方向

        Yields:
            翻译内容片段
        """
        logger.info(f"开始翻译, 方向: {direction}, 内容长度: {len(content)}")

        # 选择提示词
        if direction == "product_to_dev":
            prompt = product_to_dev.build_prompt(content)
            system_prompt = product_to_dev.SYSTEM_PROMPT
        else:
            prompt = dev_to_product.build_prompt(content)
            system_prompt = dev_to_product.SYSTEM_PROMPT

        # 调用AI生成
        async for chunk in self.ai_client.stream_generate(prompt, system_prompt):
            yield chunk

        logger.info("翻译完成")
```

### 6.6 场景识别服务模块

**文件**: `app/services/recognition.py`

**职责**:
- 场景识别逻辑
- 置信度判断
- 翻译方向建议

**实现**:

```python
import json
from app.services.ai_client import QwenClient
from app.prompts import recognition
from app.models.schemas import RecognitionResponse
from app.config import settings
from app.utils.logger import logger

class RecognitionService:
    """场景识别服务"""

    def __init__(self):
        """初始化服务"""
        self.ai_client = QwenClient()

    async def recognize(self, content: str) -> RecognitionResponse:
        """
        识别输入内容场景

        Args:
            content: 待识别内容

        Returns:
            识别结果
        """
        logger.info(f"开始场景识别, 内容长度: {len(content)}")

        # 内容太短,直接返回不确定
        if len(content) < settings.MIN_RECOGNITION_LENGTH:
            logger.warning("内容太短,无法识别")
            return RecognitionResponse(
                scene="uncertain",
                confidence=0.0,
                suggested_direction=None
            )

        # 构建提示词
        prompt = recognition.build_prompt(content)
        system_prompt = recognition.SYSTEM_PROMPT

        # 调用AI识别
        try:
            result = await self.ai_client.generate(prompt, system_prompt)
            logger.info(f"AI识别结果: {result}")

            # 解析JSON
            data = json.loads(result)
            scene = data.get("scene", "uncertain")
            confidence = float(data.get("confidence", 0.0))
            reason = data.get("reason")

            # 判断翻译方向
            threshold = settings.RECOGNITION_CONFIDENCE_THRESHOLD
            if scene == "product_requirement" and confidence > threshold:
                suggested_direction = "product_to_dev"
            elif scene == "tech_solution" and confidence > threshold:
                suggested_direction = "dev_to_product"
            else:
                suggested_direction = None

            logger.info(f"场景: {scene}, 置信度: {confidence}, 理由: {reason}, 建议方向: {suggested_direction}")

            return RecognitionResponse(
                scene=scene,
                confidence=confidence,
                reason=reason,
                suggested_direction=suggested_direction
            )

        except json.JSONDecodeError as e:
            logger.error(f"JSON解析失败: {str(e)}")
            return RecognitionResponse(
                scene="uncertain",
                confidence=0.0,
                suggested_direction=None
            )
        except Exception as e:
            logger.error(f"场景识别失败: {str(e)}")
            raise
```

```python
判断标准:
- 产品需求: 包含功能描述、用户价值、业务目标等产品视角的内容
  关键词: 需要、希望、用户、功能、体验、提升、优化(业务层面)

- 技术方案: 包含技术实现、性能指标、架构设计等开发视角的内容
  关键词: 实现了、优化了、使用、性能、QPS、响应时间、架构、算法

请分析输入内容,返回JSON格式:
{
  "scene": "product_requirement" 或 "tech_solution" 或 "uncertain",
  "confidence": 0.0-1.0,
  "reason": "判断理由(简要说明)"
}

注意:
- 如果内容太短或无法判断,返回"uncertain",confidence设为0
- confidence表示判断的确信程度,0表示完全不确定,1表示完全确定
- 只返回JSON,不要返回其他内容
"""

def build_prompt(user_input: str) -> str:
    """构建完整提示词"""
    return f"{SYSTEM_PROMPT}\n\n输入内容:\n{user_input}"
```

### 8.1 API路由实现

**文件**: `app/routers/api.py`

**完整实现**:

```python
from fastapi import APIRouter, HTTPException
from fastapi.responses import StreamingResponse
from app.models.schemas import (
    TranslateRequest,
    RecognitionRequest,
    HealthResponse
)
from app.services.translation import TranslationService
from app.services.recognition import RecognitionService
from app.config import settings
from app.utils.logger import logger
from datetime import datetime
import json

router = APIRouter(prefix="/api", tags=["API"])

# 服务实例
translation_service = TranslationService()
recognition_service = RecognitionService()

@router.post("/translate/stream")
async def translate_stream(request: TranslateRequest):
    """
    流式翻译接口

    支持SSE流式输出,实时返回翻译结果
    """
    async def generate():
        try:
            async for chunk in translation_service.translate_stream(
                request.content,
                request.direction
            ):
                data = json.dumps({
                    "type": "content",
                    "data": chunk
                }, ensure_ascii=False)
                yield f"data: {data}\n\n"

            # 完成标记
            done = json.dumps({"type": "done", "data": ""})
            yield f"data: {done}\n\n"

        except Exception as e:
            logger.error(f"翻译失败: {str(e)}")
            error = json.dumps({
                "type": "error",
                "data": f"翻译失败: {str(e)}"
            }, ensure_ascii=False)
            yield f"data: {error}\n\n"

    return StreamingResponse(
        generate(),
        media_type="text/event-stream",
        headers={
            "Cache-Control": "no-cache",
            "Connection": "keep-alive",
        }
    )

@router.post("/recognize")
async def recognize(request: RecognitionRequest):
    """
    场景识别接口

    识别输入内容属于产品需求还是技术方案
    """
    try:
        result = await recognition_service.recognize(request.content)
        return result
    except Exception as e:
        logger.error(f"识别失败: {str(e)}")
        raise HTTPException(
            status_code=500,
            detail=f"识别失败: {str(e)}"
        )

@router.get("/health")
async def health_check() -> HealthResponse:
    """
    健康检查接口

    检查服务状态和配置
    """
    return HealthResponse(
        status="ok",
        timestamp=datetime.now().isoformat(),
        api_key_configured=bool(settings.QWEN_API_KEY)
    )
```

### C. 日志工具实现

**文件**: `app/utils/logger.py`

```python
import logging
from logging.handlers import RotatingFileHandler
from app.config import settings
import os

def setup_logger():
    """设置日志"""
    # 创建日志目录
    log_dir = os.path.dirname(settings.LOG_FILE)
    if not os.path.exists(log_dir):
        os.makedirs(log_dir)

    # 创建logger
    logger = logging.getLogger("translator")
    logger.setLevel(settings.LOG_LEVEL)

    # 文件处理器
    file_handler = RotatingFileHandler(
        settings.LOG_FILE,
        maxBytes=10*1024*1024,  # 10MB
        backupCount=5
    )
    file_handler.setFormatter(
        logging.Formatter(
            '%(asctime)s - %(name)s - %(levelname)s - %(message)s'
        )
    )
    logger.addHandler(file_handler)

    # 控制台处理器
    console_handler = logging.StreamHandler()
    console_handler.setFormatter(
        logging.Formatter(
            '%(levelname)s: %(message)s'
        )
    )
    logger.addHandler(console_handler)

    return logger

logger = setup_logger()
```
