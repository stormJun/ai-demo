### 7.3 API调用模块

**文件**: `js/api.js`

```javascript
/**
 * API调用模块
 * 封装所有后端API调用
 */

const API_BASE_URL = '/api';

/**
 * 流式翻译API
 * @param {string} content - 待翻译内容
 * @param {string} direction - 翻译方向
 * @param {Function} onChunk - 接收到内容片段的回调
 * @param {Function} onDone - 翻译完成的回调
 * @param {Function} onError - 错误回调
 * @returns {Function} cancel - 用于外部取消流式请求
 */
async function translateStream(content, direction, onChunk, onDone, onError) {
    const controller = new AbortController();
    const decoder = new TextDecoder('utf-8');
    let buffer = '';

    try {
        const response = await fetch(`${API_BASE_URL}/translate/stream`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ content, direction }),
            signal: controller.signal
        });

        if (!response.ok || !response.body) {
            throw new Error(`翻译接口异常: ${response.status}`);
        }

        const reader = response.body.getReader();
        while (true) {
            const { value, done } = await reader.read();
            if (done) break;
            buffer += decoder.decode(value, { stream: true });

            // SSE切分
            const parts = buffer.split('\n\n');
            buffer = parts.pop() || '';
            parts.forEach(line => {
                if (!line.startsWith('data:')) return;
                try {
                    const payload = JSON.parse(line.replace(/^data:\s*/, ''));
                    if (payload.type === 'content') onChunk(payload.data);
                    else if (payload.type === 'done') onDone();
                    else if (payload.type === 'error') onError(payload.data);
                } catch {
                    onError('数据解析失败');
                }
            });
        }
    } catch (err) {
        if (err.name !== 'AbortError') {
            onError(err.message || '网络连接失败,请检查网络');
        }
    }

    return () => controller.abort();
}

/**
 * 场景识别API
 * @param {string} content - 待识别内容
 * @returns {Promise<Object>} 识别结果
 */
async function recognizeScene(content) {
    const url = `${API_BASE_URL}/recognize`;

    try {
        const response = await fetch(url, {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json'
            },
            body: JSON.stringify({ content })
        });

        if (!response.ok) {
            throw new Error(`HTTP error! status: ${response.status}`);
        }

        const data = await response.json();
        return data;
    } catch (error) {
        console.error('场景识别失败:', error);
        return null;
    }
}

/**
 * 健康检查API
 * @returns {Promise<Object>} 健康状态
 */
async function healthCheck() {
    const url = `${API_BASE_URL}/health`;

    try {
        const response = await fetch(url);
        const data = await response.json();
        return data;
    } catch (error) {
        console.error('健康检查失败:', error);
        return null;
    }
}
```

/**
 * Toast提示
 * @param {string} message - 提示消息
 * @param {string} type - 提示类型: info/success/warning/error
 * @param {number} duration - 显示时长(毫秒)
 */
function showToast(message, type = 'info', duration = 2000) {
    const container = document.getElementById('toastContainer');
    const toast = document.createElement('div');
    toast.className = `toast toast-${type}`;
    toast.textContent = message;

    container.appendChild(toast);

    setTimeout(() => {
        container.removeChild(toast);
    }, duration);
}

/**
 * 复制到剪贴板
 * @param {string} text - 要复制的文本
 */
function copyToClipboard(text) {
    if (navigator.clipboard && navigator.clipboard.writeText) {
        navigator.clipboard.writeText(text).then(() => {
            showToast('已复制到剪贴板', 'success');
        }).catch(() => {
            // 降级方案
            fallbackCopyToClipboard(text);
        });
    } else {
        fallbackCopyToClipboard(text);
    }
}

/**
 * 复制到剪贴板(降级方案)
 * @param {string} text - 要复制的文本
 */
function fallbackCopyToClipboard(text) {
    const textarea = document.createElement('textarea');
    textarea.value = text;
    textarea.style.position = 'fixed';
    textarea.style.opacity = '0';
    document.body.appendChild(textarea);
    textarea.select();

    try {
        document.execCommand('copy');
        showToast('已复制到剪贴板', 'success');
    } catch (err) {
        showToast('复制失败,请手动复制', 'error');
    }

    document.body.removeChild(textarea);
}

/**
 * 防抖函数
 * @param {Function} func - 要防抖的函数
 * @param {number} wait - 等待时间(毫秒)
 * @returns {Function} 防抖后的函数
 */
function debounce(func, wait) {
    let timeout;
    return function executedFunction(...args) {
        const later = () => {
            clearTimeout(timeout);
            func(...args);
        };
        clearTimeout(timeout);
        timeout = setTimeout(later, wait);
    };
}

/**
 * 节流函数
 * @param {Function} func - 要节流的函数
 * @param {number} limit - 时间限制(毫秒)
 * @returns {Function} 节流后的函数
 */
function throttle(func, limit) {
    let inThrottle;
    return function executedFunction(...args) {
        if (!inThrottle) {
            func(...args);
            inThrottle = true;
            setTimeout(() => inThrottle = false, limit);
        }
    };
}

/**
 * 字符计数
 * @param {string} text - 要计数的文本
 * @returns {number} 字符数
 */
function countChars(text) {
    return text.length;
}

    elements.translateBtn.addEventListener('click', handleTranslate);

    // 复制结果按钮
    elements.copyBtn.addEventListener('click', handleCopy);

        elements.recognitionSwitch.addEventListener('click', handleRecognitionSwitch);

/**
 * 翻译方向变化处理
 */
function handleDirectionChange(event) {
    const newDirection = event.target.value;

    // 如果有内容,询问是否切换
    if (appState.inputContent || appState.resultContent) {
        if (!confirm('切换翻译方向将清空当前内容,是否继续?')) {
            // 恢复原选项
            document.querySelector(`input[value="${appState.direction}"]`).checked = true;
            return;
        }
        handleClear();
    }

    appState.direction = newDirection;

    // 更新placeholder
    updatePlaceholder();
}

/**
 * 输入内容变化处理
 */
function handleInputChange() {
    const content = elements.inputContent.value;
    appState.inputContent = content;

    // 更新字数统计
    const count = countChars(content);
    elements.charCount.textContent = `已输入 ${count}/5000 字`;

    // 超长提示
    if (count > 5000) {
        elements.charCount.style.color = '#F5222D';
        elements.translateBtn.disabled = true;
    } else {
        elements.charCount.style.color = '#999999';
        elements.translateBtn.disabled = count === 0 || appState.isTranslating;
    }
}

/**
 * 场景识别处理(加分项)
 */
async function handleSceneRecognition() {
    const content = appState.inputContent.trim();

    // 内容太短,不识别
    if (content.length < 10) {
        elements.recognitionResult.style.display = 'none';
        return;
    }

    // 调用识别API
    const result = await recognizeScene(content);

    if (result && result.scene !== 'uncertain' && result.confidence > 0.7) {
        // 显示识别结果
        const sceneText = result.scene === 'product_requirement' ? '产品需求' : '技术方案';
        const directionText = result.suggested_direction === 'product_to_dev' ?
            '产品 → 开发' : '开发 → 产品';

        elements.recognitionText.textContent =
            `AI识别: 这是【${sceneText}】→ 翻译方向: ${directionText}`;
        elements.recognitionResult.style.display = 'flex';

        // 自动选择翻译方向
        if (result.suggested_direction) {
            document.querySelector(`input[value="${result.suggested_direction}"]`).checked = true;
            appState.direction = result.suggested_direction;
        }
    } else {
        elements.recognitionResult.style.display = 'none';
    }
}

/**
 * 清空处理
 */
function handleClear() {
    if (appState.inputContent || appState.resultContent) {
        if (!confirm('确认清空当前内容?')) {
            return;
        }
    }

    // 终止流式请求
    if (appState.cancelStream) {
        appState.cancelStream();
        appState.cancelStream = null;
        appState.isTranslating = false;
    }

    // 清空输入
    elements.inputContent.value = '';
    appState.inputContent = '';

    // 清空结果
    elements.resultContent.innerHTML = '';
    appState.resultContent = '';

    // 重置UI
    elements.charCount.textContent = '已输入 0/5000 字';
    elements.resultEmpty.style.display = 'block';
    elements.resultContent.style.display = 'none';
    elements.copyBtn.disabled = true;
    elements.translateBtn.disabled = true;
    elements.recognitionResult.style.display = 'none';

    showToast('已清空', 'info');
}

/**
 * 开始翻译处理
 */
function handleTranslate() {
    const content = appState.inputContent.trim();
    const direction = appState.direction;

    if (!content) {
        showToast('请输入内容后再翻译', 'warning');
        return;
    }

    if (content.length > 5000) {
        showToast('内容过长,请精简至5000字以内', 'error');
        return;
    }

    // 开始翻译
    startTranslation(content, direction);
}

/**
 * 开始翻译
 */
function startTranslation(content, direction) {
    // 若有正在进行的流,先中断
    if (appState.cancelStream) {
        appState.cancelStream();
        appState.cancelStream = null;
    }

    // 更新状态
    appState.isTranslating = true;
    appState.resultContent = '';

    // 更新UI
    elements.translateBtn.textContent = '翻译中...';
    elements.translateBtn.disabled = true;
    elements.resultEmpty.style.display = 'none';
    elements.resultLoading.style.display = 'flex';
    elements.resultContent.style.display = 'none';
    elements.resultContent.innerHTML = '';

    // 调用流式翻译API
    translateStream(
        content,
        direction,
        handleTranslateChunk,
        handleTranslateDone,
        handleTranslateError
    ).then(cancel => appState.cancelStream = cancel);
}

/**
 * 接收翻译片段
 */
function handleTranslateChunk(chunk) {
    // 首次接收,隐藏loading
    if (!appState.resultContent) {
        elements.resultLoading.style.display = 'none';
        elements.resultContent.style.display = 'block';
    }

    // 追加内容
    appState.resultContent += chunk;
    elements.resultContent.textContent = appState.resultContent;

    // 自动滚动到底部
    elements.resultContent.scrollTop = elements.resultContent.scrollHeight;
}

/**
 * 翻译完成
 */
function handleTranslateDone() {
    // 更新状态
    appState.isTranslating = false;
    appState.cancelStream = null;

    // 更新UI
    elements.translateBtn.textContent = '开始翻译';
    elements.translateBtn.disabled = false;
    elements.resultLoading.style.display = 'none';
    elements.resultContent.style.display = 'block';
    elements.copyBtn.disabled = false;

    showToast('翻译完成', 'success');
}

/**
 * 翻译错误处理
 */
function handleTranslateError(error) {
    // 更新状态
    appState.isTranslating = false;
    appState.cancelStream = null;

    // 更新UI
    elements.translateBtn.textContent = '开始翻译';
    elements.translateBtn.disabled = false;
    elements.resultLoading.style.display = 'none';
    elements.resultContent.style.display = 'block';
    elements.resultContent.textContent = `错误: ${error}`;

    showToast(error, 'error');
}

/**
 * 复制结果
 */
function handleCopy() {
    if (appState.resultContent) {
        copyToClipboard(appState.resultContent);
    }
}

/**
 * 识别结果切换
 */
function handleRecognitionSwitch() {
    // 切换方向
    const currentDirection = appState.direction;
    const newDirection = currentDirection === 'product_to_dev' ?
        'dev_to_product' : 'product_to_dev';

    document.querySelector(`input[value="${newDirection}"]`).checked = true;
    appState.direction = newDirection;

    // 更新提示文字
    elements.recognitionText.textContent =
        `手动选择: ${newDirection === 'product_to_dev' ? '产品 → 开发' : '开发 → 产品'}`;
}

/**
 * 更新placeholder
 */
function updatePlaceholder() {
    const placeholders = {
        'product_to_dev': '请输入产品需求描述,例如: 我们需要一个智能推荐功能...',
        'dev_to_product': '请输入技术方案说明,例如: 我们优化了数据库查询,QPS提升了30%...'
    };
    elements.inputContent.placeholder = placeholders[appState.direction];
}

 * 检查服务健康状态
 */
async function checkHealth() {
    const health = await healthCheck();
    if (health && health.status === 'ok') {
        console.log('服务正常');

    }
}

## 十、性能优化

### 10.1 代码优化

1. **防抖节流**: 场景识别使用防抖(500ms)
2. **DOM缓存**: 页面加载时缓存DOM引用
3. **事件委托**: 减少事件监听器数量

### 10.2 渲染优化

1. **流式追加**: 使用textContent追加内容
2. **自动滚动**: 只在接收内容时滚动
3. **最小重绘**: 批量更新DOM

### 10.3 资源优化

1. **无外部依赖**: 减少HTTP请求
2. **内联样式**: CSS文件合并
3. **代码压缩**: 生产环境压缩JS/CSS
